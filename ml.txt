# ============================================================

df = pd.read_json("road_accident_case_study.json")

# 1. Analyze
print(df.head())
print(df.info())
print(df.describe())

# 2. Clean
print(df.isnull().sum())

df["Weather"] = df["Weather"].fillna(
    df["Weather"].mode()[0]
)

df = df.drop_duplicates()

# 3. Feature Engineering

# Convert Date to datetime
df["Date"] = pd.to_datetime(df["Date"])

# Weekend feature
df["Weekend"] = df["Date"].dt.dayofweek >= 5

# Convert Time to hour
df["Hour"] = pd.to_datetime(
    df["Time"],
    format="%H:%M"
).dt.hour

# Time of Day
def time_of_day(hour):
    if hour < 6:
        return "Night"
    elif hour < 12:
        return "Morning"
    elif hour < 18:
        return "Afternoon"
    else:
        return "Evening"

df["Time_of_Day"] = df["Hour"].apply(time_of_day)

print(df.head())

# 4. Visualization
df["Hour"].value_counts().sort_index().plot(
    kind="line",
    marker="o"
)

plt.xlabel("Hour")
plt.ylabel("Number of Accidents")
plt.title("Accidents by Hour")
plt.grid()
plt.show()

# Q5. STUDENT PERFORMANCE - LINEAR ALGEBRA
# ============================================================

df = pd.read_csv(
    "student_performance_linear_algebra.csv"
)

# 1. Analyze
print(df.head())
print(df.info())
print(df.describe())

# 2. Clean
print(df.isnull().sum())

df = df.drop_duplicates()

num_cols = df.select_dtypes(include="number").columns
df[num_cols] = df[num_cols].fillna(
    df[num_cols].median()
)

# 3. Feature Engineering

subjects = [
    "Mathematics",
    "Physics",
    "Computer_Science",
    "English"
]

# Total marks
df["Total_Marks"] = df[subjects].sum(axis=1)

# Performance level
def performance_level(marks):
    if marks <= 50:
        return "Poor"
    elif marks <= 70:
        return "Average"
    elif marks <= 80:
        return "Good"
    else:
        return "Excellent"

df["Performance_Level"] = df[
    "Total_Marks"
].apply(performance_level)

print(
    df[
        ["Student_ID", "Total_Marks",
         "Performance_Level"]
    ].head()
)

# 4. Linear Algebra

# Convert marks to NumPy matrix
X = df[subjects].values

# Covariance matrix
cov_matrix = np.cov(
    X,
    rowvar=False
)

print("Covariance Matrix:")
print(cov_matrix)

# Eigenvalues and eigenvectors
eigenvalues, eigenvectors = np.linalg.eig(
    cov_matrix
)

print("Eigenvalues:")
print(eigenvalues)

print("Eigenvectors:")
print(eigenvectors)

# Major direction
largest = np.argmax(eigenvalues)

print(
    "Major direction:",
    eigenvectors[:, largest]
)

# Visualization
plt.bar(subjects, eigenvalues)
plt.xlabel("Subjects")
plt.ylabel("Eigenvalue")
plt.title("Eigenvalues of Student Performance")
plt.show()

# Q6. EMPLOYEE SALARY - LINEAR ALGEBRA
# ============================================================

df = pd.read_json(
    "employee_salary_linear_algebra.json"
)

# 1. Analyze
print(df.head())
print(df.info())
print(df.describe())

# 2. Clean
print(df.isnull().sum())

df = df.drop_duplicates()

num_cols = df.select_dtypes(include="number").columns
df[num_cols] = df[num_cols].fillna(
    df[num_cols].median()
)

df["Department"] = df["Department"].fillna(
    df["Department"].mode()[0]
)

# 3. Feature Engineering

# Annual salary
df["Annual_Salary"] = (
    df["Monthly_Salary"] * 12
)

# Experience level
def experience_level(years):
    if years <= 2:
        return "Beginner"
    elif years <= 5:
        return "Intermediate"
    else:
        return "Experienced"

df["Experience_Level"] = df[
    "Experience_Years"
].apply(experience_level)

print(df.head())

# 4. Linear Algebra

features = [
    "Age",
    "Experience_Years",
    "Monthly_Salary",
    "Performance_Score",
    "Working_Hours"
]

X = df[features].values

# Matrix rank
rank = np.linalg.matrix_rank(X)

print("Matrix Rank:", rank)

# Covariance matrix
cov_matrix = np.cov(
    X,
    rowvar=False
)

print("Covariance Matrix:")
print(cov_matrix)

# Eigenvalues and eigenvectors
eigenvalues, eigenvectors = np.linalg.eig(
    cov_matrix
)

print("Eigenvalues:")
print(eigenvalues)

print("Eigenvectors:")
print(eigenvectors)

# Major direction
largest = np.argmax(eigenvalues)

print(
    "Largest Eigenvalue:",
    eigenvalues[largest]
)

print(
    "Major Direction:",
    eigenvectors[:, largest]
)

# Visualization
plt.imshow(
    cov_matrix,
    cmap="viridis"
)

plt.colorbar()

plt.xticks(
    range(len(features)),
    features,
    rotation=45
)

plt.yticks(
    range(len(features)),
    features
)

plt.title("Covariance Matrix")
plt.show()
