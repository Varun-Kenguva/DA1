# Start Spark / create RDD
from pyspark.sql import SparkSession

spark = SparkSession.builder \
    .appName("BDA_Lab") \
    .master("local[*]") \
    .getOrCreate()

data = [1, 2, 3, 4, 5]

rdd = spark.sparkContext.parallelize(data)

print("RDD Data:")
print(rdd.collect())

print("RDD Sum:")
print(rdd.sum())


# ==========================================
# 5. PYSPARK DATAFRAME
# ==========================================

# Create sample textbook data
data = [
    (1, "Rahul", 85),
    (2, "Priya", 90),
    (3, "Amit", 78)
]

columns = ["ID", "Name", "Marks"]

# Create DataFrame
df = spark.createDataFrame(data, columns)

# Display Data
print("DataFrame:")
df.show()

# Display Schema
print("Schema:")
df.printSchema()

# Number of rows
print("Number of rows:", df.count())

# Stop Spark
spark.stop()
